cmake_minimum_required(VERSION 3.16)
project(liuo_tunnel_sim)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(liuo_tunnel REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(liuo_tunnel_msgs REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
add_library(liuo_tunnel_renderer SHARED src/first_hit_lidar_renderer.cpp)
target_include_directories(liuo_tunnel_renderer PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
target_link_libraries(liuo_tunnel_renderer PUBLIC ${PCL_LIBRARIES})
install(TARGETS liuo_tunnel_renderer LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_physics test/test_physics.cpp)
 target_link_libraries(test_physics liuo_tunnel_renderer)
 ament_add_gtest(test_renderer test/test_renderer.cpp)
 target_link_libraries(test_renderer liuo_tunnel_renderer)
endif()
add_executable(liuo_rail_sim src/rail_sim.cpp)
target_link_libraries(liuo_rail_sim liuo_tunnel_renderer liuo_tunnel::liuo_tunnel_core)
ament_target_dependencies(liuo_rail_sim rclcpp sensor_msgs nav_msgs rosgraph_msgs liuo_tunnel_msgs)
install(TARGETS liuo_rail_sim DESTINATION lib/${PROJECT_NAME})
ament_package()
