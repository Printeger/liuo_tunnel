cmake_minimum_required(VERSION 3.16)
project(liuo_tunnel)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(liuo_tunnel_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(GTSAM REQUIRED)
find_package(gtsam_points REQUIRED)
if(NOT GTSAM_POINTS_USE_CUDA)
 message(FATAL_ERROR "GPU baseline requires CUDA-enabled gtsam_points")
endif()
find_package(spdlog REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem)
find_package(OpenMP REQUIRED)
get_filename_component(LIUO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_library(liuo_tunnel_core SHARED src/optimizers/isam2_ext.cpp src/optimizers/isam2_ext_impl.cpp src/optimizers/incremental_fixed_lag_smoother_ext.cpp src/util/config.cpp src/util/run_log_manager.cpp src/util/logging.cpp src/common/imu_integration.cpp src/common/cloud_covariance_estimation.cpp)
target_include_directories(liuo_tunnel_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(liuo_tunnel_core PRIVATE LIUO_SOURCE_ROOT="${LIUO_ROOT}" LIUO_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
target_link_libraries(liuo_tunnel_core PUBLIC gtsam gtsam_points::gtsam_points spdlog::spdlog Boost::filesystem OpenMP::OpenMP_CXX)
install(TARGETS liuo_tunnel_core EXPORT liuo_tunnel_targets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(liuo_tunnel_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 GTSAM gtsam_points spdlog Boost OpenMP)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 foreach(test test_run_log_manager test_run_log_manager_standalone test_imu test_vgicp test_uwb)
  ament_add_gtest(${test} test/${test}.cpp)
  target_link_libraries(${test} liuo_tunnel_core)
 endforeach()
endif()
add_executable(liuo_tunnel_node apps/liuo_tunnel_node.cpp)
target_link_libraries(liuo_tunnel_node liuo_tunnel_core CUDA::cudart)
ament_target_dependencies(liuo_tunnel_node rclcpp sensor_msgs nav_msgs diagnostic_msgs liuo_tunnel_msgs)
install(TARGETS liuo_tunnel_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
